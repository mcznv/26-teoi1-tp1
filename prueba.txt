  Prueba de variables, tipos de datos, ciclos, if y setswitch
  //* Comentario anidado de 1 nivel segun consigna *//
*//

DECLARE.SECTION
    [a, b, c] ::= [INT, FLOAT, INT]
    [texto] ::= [STRING]
ENDDECLARE.SECTION

PROGRAM.SECTION
    //* Asignaciones simples y formatos de reales *//
    a ::= 5 + 10
    b ::= 3.14 * 2.0
    texto ::= "Hola Compilador"

    //* Ciclo WHILE con conector logico AND *//
    WHILE (a < 20 AND b >= 1.5)
        a ::= a + 1
    ENDWHILE

    //* Condicional IF con ELSE *//
    IF (a == 20)
        write "Fin del ciclo"
    ELSE
        write 0
    ENDIF

    //* Prueba del Tema Especial: SETSWITCH *//
    SETSWITCH (a + c * 2)
    CASE 10:
        c ::= 200
        a ::= 100
    CASE 22:
        c ::= a + 50
    CASE 30:
        c ::= a
    ELSECASE:
        a ::= 0
    ENDSETCASE

    write a
ENDPROGRAM.SECTION
